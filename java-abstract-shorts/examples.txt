// [1. 왜 필요할까] Main.java → 출력:
//   서준 안녕
public class Main {
  public static void main(String[] args) {
    Person p =
        new Person("서준");
    p.hello();
  }
}

class Person {
  String name;
  Person(String name) {
    this.name = name;
  }
  void hello() {
    System.out.println(
        name + " 안녕");
  }
}

// [1. 추상 메소드] Main.java → 컴파일 성공 확인 (추상 메소드가 있는 추상 클래스)
public class Main {
  public static void main(String[] args) {
  }
}

abstract class Person {
  String name;
  Person(String name) {
    this.name = name;
  }
  abstract void hello();
  void bye() {
    System.out.println(
        name + " 잘 가");
  }
}

class Student extends Person {
  int score;
  Student(String name,
          int score) {
    super(name);
    this.score = score;
  }
  @Override
  void hello() {
    System.out.println(
        "학생 " + name + " 안녕");
  }
}

class Teacher extends Person {
  Teacher(String name) {
    super(name);
  }
  @Override
  void hello() {
    System.out.println(
        name + " 선생님 안녕");
  }
}
