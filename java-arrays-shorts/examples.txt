// 각 예제를 파일 이름대로 저장한 뒤: javac -encoding UTF-8 Main.java && java Main

// [1. 배열이란] Main.java → 출력:
//   90
//   75
//   88
public class Main {
  public static void main(String[] args) {
    int[] scores = {90, 75, 88};
    System.out.println(scores[0]);
    System.out.println(scores[1]);
    System.out.println(scores[2]);
  }
}

// [2. new로 만들기] Main.java → 출력:
//   0
//   0.0
//   false
//   null
public class Main {
  public static void main(String[] args) {
    System.out.println((new int[1])[0]);
    System.out.println((new double[1])[0]);
    System.out.println((new boolean[1])[0]);
    System.out.println((new String[1])[0]);
  }
}

// [5. 다차원 배열] Main.java → 출력:
//   2
//   3
//   000007
public class Main {
  public static void main(String[] args) {
    int[][] seats = new int[2][3];
    seats[1][2] = 7;
    System.out.println(seats.length);
    System.out.println(seats[0].length);
    System.out.print(seats[0][0]);
    System.out.print(seats[0][1]);
    System.out.print(seats[0][2]);
    System.out.print(seats[1][0]);
    System.out.print(seats[1][1]);
    System.out.print(seats[1][2]);
  }
}

  }
