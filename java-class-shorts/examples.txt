// [3-1. 필드의 기본값] Main.java → 출력:
//   0 0.0 false null
public class Main {
  public static void main(String[] args) {
    D d = new D();
    System.out.println(d.i + " " + d.d + " " + d.b + " " + d.s);
  }
}

class D { int i; double d; boolean b; String s; }

    Student s1 = new Student();

    System.out.println(s1.score);
  }
}
