// 각 예제를 파일 이름대로 저장한 뒤: javac -g -encoding UTF-8 Main.java && java Main
// 확인한 JDK: openjdk version "25.0.4.1" 2026-08-18

// [도입] Main.java → 실행 오류: Exception in thread "main" java.lang.ArithmeticException: / by zero
public class Main {
  public static void main(String[] args) {
    int a = 10 / 0;
    System.out.println(a);
    System.out.println("끝");
  }
}


// [1. try-catch] Main.java → 출력:
//   0으로 나눌 수 없어요
//   끝
public class Main {
  public static void main(String[] args) {
    try {
      int a = 10 / 0;
      System.out.println(a);
    } catch (ArithmeticException e) {
      System.out.println(
          "0으로 나눌 수 없어요");
    }
    System.out.println("끝");
  }
}


// [2. 여러 catch] Main.java → 출력:
//   For input string: "abc"
//   Index 3 out of bounds for length 3
public class Main {
  public static void main(String[] args) {
    int[] scores = {85, 92, 78};
    try {
      Integer.parseInt("abc");
    } catch (
      NumberFormatException e) {
      System.out.println(
        e.getMessage());
    }
    try {
      int n = scores[3];
    } catch (
      ArrayIndexOutOfBoundsException
        e) {
      System.out.println(
        e.getMessage());
    }
  }
}


// [2-1. catch 순서 함정] Main.java → 컴파일 오류: exception ArithmeticException has already been caught
public class Main {
  public static void main(String[] args) {
    try {
      int a = 10 / 0;
    } catch (Exception e) {
      System.out.println("예외");
    } catch (ArithmeticException e) {
      System.out.println("0 나누기");
    }
  }
}


// [3. finally] Main.java → 출력:
//   85
//   끝
public class Main {
  public static void main(String[] args) {
    String s = "85";
    try {
      int n = Integer.parseInt(s);
      System.out.println(n);
    } catch (
      NumberFormatException e) {
      System.out.println("숫자 아님");
    } finally {
      System.out.println("끝");
    }
  }
}
