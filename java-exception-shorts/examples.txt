// 각 예제를 파일 이름대로 저장한 뒤: javac -g -encoding UTF-8 Main.java && java Main
// 확인한 JDK: openjdk version "25.0.4.1" 2026-08-18

// [도입] Main.java → 실행 오류: Exception in thread "main" java.lang.ArithmeticException: / by zero
public class Main {
  public static void main(String[] args) {
    int a = 10 / 0;
    System.out.println(a);
    System.out.println("끝");
  }
}


// [1. try-catch] Main.java → 출력:
//   0으로 나눌 수 없어요
//   끝
public class Main {
  public static void main(String[] args) {
    try {
      int a = 10 / 0;
      System.out.println(a);
    } catch (ArithmeticException e) {
      System.out.println(
          "0으로 나눌 수 없어요");
    }
    System.out.println("끝");
  }
}


// [4. throw] Main.java → 출력:
//   점수 오류: 120
public class Main {
  static void checkScore(
      int score) {
    if (score < 0 || score > 100)
      throw new
        IllegalArgumentException(
          "점수 오류: " + score);
  }

  public static void main(String[] args) {
    
    try {
      checkScore(120);
    } catch (
      IllegalArgumentException e) {
      System.out.println(
          e.getMessage());
    }
  }
}
