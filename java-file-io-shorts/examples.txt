// [3. 읽어서 계산] Main.java → 출력:
//   255
// 실행 뒤 scores.txt:
//   민수,85
//   서준,92
//   지우,78
import java.io.*;
import java.nio.file.*;
import java.util.*;

public class Main {
  public static void main(String[] args) throws IOException {
    try (BufferedWriter w =
      new BufferedWriter(
        new FileWriter(
          "scores.txt"))) {
      w.write("민수,85");
      w.newLine();
      w.write("서준,92");
      w.newLine();
    }
    try (BufferedWriter w =
      new BufferedWriter(
        new FileWriter(
          "scores.txt", true))) {
      w.write("지우,78");
      w.newLine();
    }
    int sum = 0;
    try (BufferedReader r =
      new BufferedReader(
        new FileReader(
          "scores.txt"))) {
      String line;
      while ((line = r.readLine())
             != null) {
        String[] p = line.split(",");
        sum += Integer.parseInt(
          p[1]);
      }
    }
    System.out.println(sum);
  }
}
