// 영상에 나오는 예제 프로그램 (build.py가 생성 · 직접 고치지 말 것)
// 각 예제를 빈 폴더에 파일 이름대로 저장한 뒤: javac -g -encoding UTF-8 Main.java && java Main
// (scores.txt는 실행한 폴더에 생긴다. 앞 장면의 파일 상태는 각 예제 안에서 먼저 만든다)
// 확인한 JDK: openjdk version "25.0.4.1" 2026-08-18

// [1. 쓰기] Main.java → 출력 없음 (컴파일·실행 성공)
// 실행 뒤 scores.txt:
//   민수,85
//   서준,92
import java.io.*;
import java.nio.file.*;
import java.util.*;

public class Main {
  public static void main(String[] args) throws IOException {
    try (BufferedWriter w =
      new BufferedWriter(
        new FileWriter(
          "scores.txt"))) {
      w.write("민수,85");
      w.newLine();
      w.write("서준,92");
      w.newLine();
    }
  }
}
