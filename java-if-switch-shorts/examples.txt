// [4. switch문] Main.java → 출력:
//   수요일
public class Main {
  public static void main(String[] args) {
    int score = 85;
    int day = 3;
    switch (day) {
      case 3:
        System.out.println("수요일");
        break;
      case 4:
        System.out.println("목요일");
        break;
      default:
        System.out.println("그 외");
    }
  }
}

// [2. if-else] Main.java → 출력:
//   다음 기회
public class Main {
  public static void main(String[] args) {
    int score = 85;
    int day = 3;
    if (score >= 90) {
      System.out.println("장학금");
    } else {
      System.out.println("다음 기회");
    }
  }
}

// [3. else if] Main.java → 출력:
//   B
public class Main {
  public static void main(String[] args) {
    int score = 85;
    int day = 3;
    if (score >= 90) {
      System.out.println("A");
    } else if (score >= 80) {
      System.out.println("B");
    } else if (score >= 70) {
      System.out.println("C");
    } else {
      System.out.println("F");
    }
  }
}

// [5. 화살표 switch] Main.java → 출력:
//   수요일
public class Main {
  public static void main(String[] args) {
    int score = 85;
    int day = 3;
    String name = switch (day) {
      case 3 -> "수요일";
      case 4 -> "목요일";
      default -> "그 외";
    };
    System.out.println(name);
  }
}

// [5. 화살표 switch] Main.java → 출력:
//   평일
public class Main {
  public static void main(String[] args) {
    int score = 85;
    int day = 3;
    String type = switch (day) {
      case 6, 7 -> "주말"; // 묶기
      default -> "평일";
    };
    System.out.println(type);
  }
}
