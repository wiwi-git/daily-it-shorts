// [4. super()] Main.java → 출력:
//   부모 생성자
//   자식 생성자
public class Main {
  public static void main(String[] args) {
    new Student("민수", 85);
  }
}

class Person {
  String name;
  Person(String name) {
    System.out.println("부모 생성자");
    this.name = name;
  }
}

class Student extends Person {
  int score;
  Student(String name,
          int score) {
    super(name);
    System.out.println("자식 생성자");
    this.score = score;
  }
}
