// [1. 선언] Main.java → 컴파일 성공 확인 (인터페이스 Studyable 선언)
public class Main {
  public static void main(String[] args) {
  }
}

abstract class Person {
  String name;
  Person(String name) {
    this.name = name;
  }
  abstract void hello();
  void bye() {
    System.out.println(
        name + " 잘 가");
  }
}

interface Studyable {
  int MAX_HOURS = 8;
  void study();
  default void rest() {
    System.out.println(
        "쉬는 중");
  }
}

interface Teachable {
  void teach();
}

class Student extends Person
    implements Studyable {
  int score;
  Student(String name,
          int score) {
    super(name);
    this.score = score;
  }
  @Override
  void hello() {
    System.out.println(
        "학생 " + name + " 안녕");
  }
  @Override
  public void study() {
    System.out.println(
        name + " 공부 중");
  }
}

class Teacher extends Person
    implements Teachable, Studyable {
  Teacher(String name) {
    super(name);
  }
  @Override
  void hello() {
    System.out.println(
        name + " 선생님 안녕");
  }
  @Override
  public void teach() {
    System.out.println(
        name + " 수업 중");
  }
  @Override
  public void study() {
    System.out.println(
        name + " 공부 중");
  }
}
