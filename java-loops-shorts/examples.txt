// 영상에 나오는 예제 프로그램 (build.py가 생성 · 직접 고치지 말 것)
// 각 예제를 파일 이름대로 저장한 뒤: javac -encoding UTF-8 Main.java && java Main

// [2. 합계 구하기] Main.java → 출력:
//   1 3 6 10 15 21 28 36 45 55
public class Main {
  public static void main(String[] args) {
    int sum = 0;
    for (int i = 1; i <= 10; i++) {
      sum += i;
      System.out.print(sum + " ");
    }
  }
}

// [3. while문] Main.java → 출력:
//   1 2 4 8 16 32 64 128
public class Main {
  public static void main(String[] args) {
    int n = 1;
    System.out.print(n + " ");
    while (n <= 100) {
      n *= 2;
      System.out.print(n + " ");
    }
  }
}

// [2-1. 배열과 for문] Main.java → 출력:
//   85 92 78
public class Main {
  public static void main(String[] args) {
    int[] arr = {85, 92, 78};
    for (int i = 0; i < arr.length;
         i++) {
      System.out.print(arr[i] + " ");
    }
  }
}

// [2-1. 배열과 for문] Main.java → 출력:
//   85 92 78
public class Main {
  public static void main(String[] args) {
    int[] arr = {85, 92, 78};
    for (int x : arr) {
      System.out.print(x + " ");
    }
  }
}

// [3. while문] Main.java → 출력:
//   128
public class Main {
  public static void main(String[] args) {
    int n = 1;
    while (n <= 100) {
      n *= 2;
    }
    System.out.println(n);
  }
}

// [3-1. do-while문] Main.java → 출력:
//   200
public class Main {
  public static void main(String[] args) {
    int n = 200;
    do {
      System.out.println(n);
    } while (n <= 100);
  }
}

// [3-1. do-while문] Main.java → 출력 없음 (컴파일·실행 성공)
public class Main {
  public static void main(String[] args) {
    int n = 200;
    while (n <= 100) {
      System.out.println(n);
    }
  }
}

// [4. break와 continue] Main.java → 출력:
//   2 4 6
public class Main {
  public static void main(String[] args) {
    for (int i = 1; i <= 10; i++) {
      if (i % 2 == 1) continue;
      if (i > 6) break;
      System.out.print(i + " ");
    }
  }
}

// [4-1. 무한 루프 주의] Main.java → 컴파일만 확인 (실행하지 않음: n이 그대로 1이라 끝나지 않음)
public class Main {
  public static void main(String[] args) {
    int n = 1;
    while (n <= 100) {
      System.out.println(n);
    }
  }
}

// [4-1. 무한 루프 주의] Main.java → 출력:
//   1
//   2
//   4
//   8
//   16
//   32
//   64
public class Main {
  public static void main(String[] args) {
    int n = 1;
    while (n <= 100) {
      System.out.println(n);
      n *= 2; // ← 꼭 필요
    }
  }
}
