// [5-1. 기본 생성자 주의] Main.java → 출력:
//   null 민수
public class Main {
  public static void main(String[] args) {
    Student a = new Student();
    Student b = new Student("민수", 85);
    System.out.println(a.name + " " + b.name);
  }
}

class Student {
  String name; // 이름
  int score;   // 점수
  Student() {} // 빈 괄호용
  Student(String name,
          int score) {
    this.name = name;
    this.score = score;
  }
}

// [2. 매개변수] Main.java → 출력:
//   90
public class Main {
  public static void main(String[] args) {
    Student s1 = new Student();
    s1.name = "민수";
    s1.score = 85;
    s1.addScore(5);
    System.out.println(s1.score);
  }
}

class Student {
  String name; // 이름
  int score;   // 점수
  void addScore(int n) {
    score += n;
  }
}

// [3. 반환값 return] Main.java → 출력:

public class Main {
  public static void main(String[] args) {
    Student s1 = new Student();
    s1.name = "민수";
    s1.score = 85;
    System.out.println(s1.isPass());
  }
}

class Student {
  String name; // 이름
  int score;   // 점수
  boolean isPass() {
    return score >= 60;
  }
}
