// [1. 부모 타입에 자식 객체] Main.java → 출력 없음 (컴파일·실행 성공)
public class Main {
  public static void main(String[] args) {
    Person p = new Student(
        "민수", 85);
  }
}

abstract class Person {
  String name;
  Person(String name) {
    this.name = name;
  }
  abstract void hello();
  void bye() {
    System.out.println(
        name + " 잘 가");
  }
}

interface Studyable {
  int MAX_HOURS = 8;
  void study();
  default void rest() {
    System.out.println(
        "쉬는 중");
  }
}

interface Teachable {
  void teach();
}

class Student extends Person
    implements Studyable {
  int score;
  Student(String name,
          int score) {
    super(name);
    this.score = score;
  }
  @Override
  void hello() {
    System.out.println(
        "학생 " + name + " 안녕");
  }
  @Override
  public void study() {
    System.out.println(
        name + " 공부 중");
  }
}

class Teacher extends Person
    implements Teachable, Studyable {
  Teacher(String name) {
    super(name);
  }
  @Override
  void hello() {
    System.out.println(
        name + " 선생님 안녕");
  }
  @Override
  public void teach() {
    System.out.println(
        name + " 수업 중");
  }
  @Override
  public void study() {
    System.out.println(
        name + " 공부 중");
  }
}
