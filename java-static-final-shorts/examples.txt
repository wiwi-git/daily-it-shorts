// [4. final 메소드] Main.java → 컴파일 오류: hello() in Student cannot override hello() in Person
public class Main {
  public static void main(String[] args) {
  }
}

class Person {
  String name;
  final void hello() {
    System.out.println(
        "안녕하세요");
  }
}

class Student extends Person {
  void hello() {
    System.out.println(
        "안녕하세요");
  }
}

// [5. static final 상수] Main.java → 출력:
//   true
public class Main {
  public static void main(String[] args) {
    Student s = new Student(
        "민수", 85);
    System.out.println(
        s.isPass());
  }
}

class Student {
  static final int
      PASS_SCORE = 60;
  final String name;
  int score;
  static int count;
  Student(String name,
          int score) {
    this.name = name;
    this.score = score;
    count++;
  }
  static int getCount() {
    return count;
  }
  boolean isPass() {
    return score >= PASS_SCORE;
  }
}
