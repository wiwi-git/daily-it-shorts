// [1. 스레드 만들기] Main.java → 출력 예시 (20회 실행해 확인 · 실행할 때마다 순서·값이 다를 수 있음):
//   민수 85
//   main 끝
//   서준 92
public class Main {
  public static void main(String[] args) {
    Thread a = new Thread(() ->
      System.out.println("민수 85"));
    Thread b = new Thread(() ->
      System.out.println("서준 92"));
    a.start();
    b.start();
    System.out.println("main 끝");
  }
}

// [2. sleep과 join] Main.java → 출력 예시 (5회 실행해 확인 · 실행할 때마다 순서·값이 다를 수 있음):
//   채점 끝
//   민수 85
public class Main {
  public static void main(String[] args) throws InterruptedException {
    Thread t = new Thread(() -> {
      try {
        Thread.sleep(1000);
      } catch (
        InterruptedException e) {}
      System.out.println("민수 85");
    });
    t.start();
    System.out.println("채점 끝");
  }
}


// [2. sleep과 join] Main.java → 출력 예시 (5회 실행해 확인 · 실행할 때마다 순서·값이 다를 수 있음):
//   민수 85
//   채점 끝
public class Main {
  public static void main(String[] args) throws InterruptedException {
    Thread t = new Thread(() -> {
      try {
        Thread.sleep(1000);
      } catch (
        InterruptedException e) {}
      System.out.println("민수 85");
    });
    t.start();
    t.join();
    System.out.println("채점 끝");
  }
}


// [3. 경쟁 조건] Main.java → 출력 예시 (20회 실행해 확인 · 실행할 때마다 순서·값이 다를 수 있음):
//   13426
public class Main {
  static int count = 0;
  static void inc() {
    for (int i = 0; i < 10000; i++)
      count++;
  }
  

  public static void main(String[] args) throws InterruptedException {
    Thread a = new Thread(Main::inc);
    Thread b = new Thread(Main::inc);
    a.start(); b.start();
    a.join(); b.join();
    System.out.println(count);
  }
}
