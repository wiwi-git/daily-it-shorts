// [3. 자바는 왜?] Main.java → 출력:
//   1 2 1
import java.util.*;
import java.util.function.*;

public class Main {
  static IntSupplier
      makeCounter() {
    int[] count = {0};
    return () -> ++count[0];
  }

  public static void main(String[] args) {
    IntSupplier c1 = makeCounter();
    IntSupplier c2 = makeCounter();
    System.out.println(c1.getAsInt() + " " + c1.getAsInt() + " " + c2.getAsInt());
  }
}

//   0
//   1
//   2
import java.util.*;
import java.util.function.*;

public class Main {
  public static void main(String[] args) {
    List<IntSupplier> list = new ArrayList<>();
    for (int n : new int[]{0, 1, 2})
      list.add(() -> n);
    for (IntSupplier s : list) System.out.println(s.getAsInt());
  }
}
